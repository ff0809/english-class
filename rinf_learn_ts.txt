import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from collections import deque
import random

# Deep Q-Network for stock trading
class DQNAgent:
    def __init__(self, state_size, action_size):
        self.state_size = state_size
        self.action_size = action_size  # 0: Hold, 1: Buy, 2: Sell
        self.memory = deque(maxlen=2000)
        self.gamma = 0.95  # discount rate
        self.epsilon = 1.0  # exploration rate
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
        self.learning_rate = 0.001
        
        # Simple neural network weights (2 hidden layers)
        self.w1 = np.random.randn(state_size, 64) * 0.1
        self.b1 = np.zeros(64)
        self.w2 = np.random.randn(64, 32) * 0.1
        self.b2 = np.zeros(32)
        self.w3 = np.random.randn(32, action_size) * 0.1
        self.b3 = np.zeros(action_size)
    
    def relu(self, x):
        return np.maximum(0, x)
    
    def predict(self, state):
        """Forward pass through the network"""
        h1 = self.relu(np.dot(state, self.w1) + self.b1)
        h2 = self.relu(np.dot(h1, self.w2) + self.b2)
        output = np.dot(h2, self.w3) + self.b3
        return output
    
    def act(self, state):
        """Choose action using epsilon-greedy policy"""
        if np.random.rand() <= self.epsilon:
            return random.randrange(self.action_size)
        act_values = self.predict(state)
        return np.argmax(act_values)
    
    def remember(self, state, action, reward, next_state, done):
        """Store experience in replay memory"""
        self.memory.append((state, action, reward, next_state, done))
    
    def replay(self, batch_size):
        """Train on a batch of experiences"""
        if len(self.memory) < batch_size:
            return
        
        minibatch = random.sample(self.memory, batch_size)
        
        for state, action, reward, next_state, done in minibatch:
            target = reward
            if not done:
                target = reward + self.gamma * np.max(self.predict(next_state))
            
            # Forward pass
            h1 = self.relu(np.dot(state, self.w1) + self.b1)
            h2 = self.relu(np.dot(h1, self.w2) + self.b2)
            output = np.dot(h2, self.w3) + self.b3
            
            # Backward pass (simplified gradient descent)
            target_f = output.copy()
            target_f[action] = target
            
            # Calculate gradients and update weights
            d_output = (output - target_f) * self.learning_rate
            
            dw3 = np.outer(h2, d_output)
            db3 = d_output
            
            d_h2 = np.dot(d_output, self.w3.T)
            d_h2[h2 <= 0] = 0  # ReLU derivative
            
            dw2 = np.outer(h1, d_h2)
            db2 = d_h2
            
            d_h1 = np.dot(d_h2, self.w2.T)
            d_h1[h1 <= 0] = 0
            
            dw1 = np.outer(state, d_h1)
            db1 = d_h1
            
            # Update weights
            self.w3 -= dw3
            self.b3 -= db3
            self.w2 -= dw2
            self.b2 -= db2
            self.w1 -= dw1
            self.b1 -= db1
        
        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay


# Trading Environment
class TradingEnvironment:
    def __init__(self, data, initial_balance=10000):
        self.data = data
        self.initial_balance = initial_balance
        self.reset()
    
    def reset(self):
        self.balance = self.initial_balance
        self.shares_held = 0
        self.current_step = 0
        self.total_profit = 0
        self.trades = []
        return self._get_state()
    
    def _get_state(self):
        """Get current state representation"""
        if self.current_step >= len(self.data) - 5:
            return np.zeros(10)
        
        # State: [price, 5-day changes, volume, balance ratio, shares held]
        price = self.data[self.current_step]
        price_changes = []
        for i in range(1, 6):
            if self.current_step - i >= 0:
                price_changes.append(
                    (self.data[self.current_step] - self.data[self.current_step - i]) / 
                    self.data[self.current_step - i]
                )
            else:
                price_changes.append(0)
        
        state = np.array([
            price / 1000,  # normalized price
            *price_changes,
            self.balance / self.initial_balance,
            self.shares_held / 10,  # normalized shares
            0.5,  # placeholder for volume
            self.current_step / len(self.data)
        ])
        return state
    
    def step(self, action):
        """Execute action and return next state, reward, done"""
        current_price = self.data[self.current_step]
        done = self.current_step >= len(self.data) - 1
        
        reward = 0
        
        # Action: 0=Hold, 1=Buy, 2=Sell
        if action == 1:  # Buy
            shares_to_buy = self.balance // current_price
            if shares_to_buy > 0:
                cost = shares_to_buy * current_price
                self.balance -= cost
                self.shares_held += shares_to_buy
                self.trades.append(('BUY', self.current_step, current_price, shares_to_buy))
                reward = -0.01  # small penalty for transaction
        
        elif action == 2:  # Sell
            if self.shares_held > 0:
                revenue = self.shares_held * current_price
                profit = revenue - (self.shares_held * self._get_avg_buy_price())
                self.balance += revenue
                self.trades.append(('SELL', self.current_step, current_price, self.shares_held))
                self.shares_held = 0
                reward = profit / self.initial_balance  # normalized profit
        
        # Move to next step
        self.current_step += 1
        
        # Calculate portfolio value
        next_state = self._get_state()
        portfolio_value = self.balance + (self.shares_held * self.data[min(self.current_step, len(self.data)-1)])
        self.total_profit = portfolio_value - self.initial_balance
        
        return next_state, reward, done
    
    def _get_avg_buy_price(self):
        """Calculate average buy price from trade history"""
        buy_trades = [t for t in self.trades if t[0] == 'BUY']
        if not buy_trades:
            return self.data[self.current_step]
        total_cost = sum(t[2] * t[3] for t in buy_trades)
        total_shares = sum(t[3] for t in buy_trades)
        return total_cost / total_shares if total_shares > 0 else self.data[self.current_step]


# Generate synthetic stock price data
def generate_stock_data(days=252, initial_price=100):
    """Generate synthetic stock prices with trend and noise"""
    np.random.seed(42)
    prices = [initial_price]
    for _ in range(days - 1):
        change = np.random.normal(0.001, 0.02)  # 0.1% drift, 2% volatility
        new_price = prices[-1] * (1 + change)
        prices.append(max(new_price, 1))  # prevent negative prices
    return np.array(prices)


# Training loop
def train_agent(episodes=50):
    stock_data = generate_stock_data(days=252)
    env = TradingEnvironment(stock_data)
    state_size = 10
    action_size = 3
    agent = DQNAgent(state_size, action_size)
    batch_size = 32
    
    profits = []
    
    for e in range(episodes):
        state = env.reset()
        total_reward = 0
        
        for time in range(len(stock_data) - 1):
            action = agent.act(state)
            next_state, reward, done = env.step(action)
            agent.remember(state, action, reward, next_state, done)
            state = next_state
            total_reward += reward
            
            if done:
                profits.append(env.total_profit)
                print(f"Episode {e+1}/{episodes}, Profit: ${env.total_profit:.2f}, "
                      f"Epsilon: {agent.epsilon:.3f}, Total Reward: {total_reward:.3f}")
                break
            
            if len(agent.memory) > batch_size:
                agent.replay(batch_size)
    
    return agent, env, profits


# Run training
print("Training RL agent for stock trading...")
print("=" * 60)
trained_agent, final_env, profit_history = train_agent(episodes=50)
